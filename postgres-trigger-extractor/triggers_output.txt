CREATE TRIGGER after_delete_trigger AFTER DELETE ON public.sample_table FOR EACH ROW EXECUTE FUNCTION log_delete()

Function Name: log_delete

Function Definition:
CREATE OR REPLACE FUNCTION public.log_delete()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
    INSERT INTO trigger_log (table_name, operation, old_data, new_data, changed_at)
    VALUES (TG_TABLE_NAME, 'DELETE', row_to_json(OLD), NULL, NOW());
    RETURN OLD;
END;
